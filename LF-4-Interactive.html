#Import Data

 LFdata <- read_excel("LF_July_22.xlsx")

df <- LFdata

# Create Age groups
df <- df %>%
  mutate(
    Age_group = case_when(
      age < 10 ~ "<10",
      age >=10 & age <=19 ~ "10-19",
      age >=20 & age <=29 ~ "20-29",
      age >=30 & age <=40 ~ "30-40",
      age >40 ~ ">40",
      TRUE ~ NA_character_
    )
  )

Select Indicators

selectInput(
  "site",
  "Site",
  choices = c("All", sort(unique(df$site))),
  selected = "All"
)
selectInput(
  "sex",
  "Sex",
  choices = c("All", sort(unique(df$sex))),
  selected = "All"
)
selectInput(
  "age",
  "Age Group",
  choices = c("All", sort(unique(df$Age_group))),
  selected = "All"
)
filtered <- reactive({

x <- df

if(input$site!="All")
  x <- filter(x, site==input$site)

if(input$sex!="All")
  x <- filter(x, sex==input$sex)

if(input$age!="All")
  x <- filter(x, Age_group==input$age)

x

})

Prevalence

Interactive Prevalence

renderPlotly({

prev <- filtered() %>%

group_by(site,
         sex,
         Age_group) %>%

summarise(

Prev=
mean(
fts_test_result=="Positive",
na.rm=TRUE
)*100,

.groups="drop"

)

p <- ggplot(
  prev,
  aes(
    Age_group,
    Prev,
    fill = sex,
    text = paste(
      "Age:", Age_group,
      "<br>Sex:", sex,
      "<br>Prev:", round(Prev, 1), "%"
    )
  )
) +
  geom_col(position = "dodge") +
  scale_fill_manual(values = c(
    "Male" = "#1f77b4",    # blue
    "Female" = "#e63946"   # red
  )) +
  facet_wrap(~site) +
  theme_bw()

ggplotly(p, tooltip = "text")
})
